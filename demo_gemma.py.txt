import torchfrom sos_modulator import SoSAttentionModulatordef run_stability_test():    print("=== Initialisation de la Sandbox SoS (Gemma Core Emulation) ===")        # Paramètres d'une couche classique (Gemma 2B)    batch_size = 1    num_heads = 8    seq_len = 32    d_key = 256  # Dimension des clés        # 1. Instanciation de la boussole SoS    sos_filter = SoSAttentionModulator(d_key=d_key, gamma=1.2)        # 2. Simulation d'un signal latent chaotique (Haute Anergie / Bruit)    print("\n[1] Génération d'un flux informationnel instable...")    Q = torch.randn(batch_size, num_heads, seq_len, d_key) * 5.0  # Forte variance    K = torch.randn(batch_size, num_heads, seq_len, d_key) * 5.0    V = torch.randn(batch_size, num_heads, seq_len, d_key)        # 3. Calcul classique vs Calcul filtré SoS    # Simulation du produit de corrélation standard    scores_standard = torch.matmul(Q, K.transpose(-2, -1)) / (d_key ** 0.5)    probs_standard = torch.softmax(scores_standard, dim=-1)    entropy_standard = -torch.sum(probs_standard * torch.log(probs_standard + 1e-9)) / (batch_size * num_heads)        # Application du filtre SoS    output_sos, probs_sos = sos_filter(Q, K, V)    entropy_sos = -torch.sum(probs_sos * torch.log(probs_sos + 1e-9)) / (batch_size * num_heads)        # 4. Affichage des métriques de stabilité    print("\n=== RÉSULTATS DE LA SIMULATION THERMODYNAMIQUE ===")    print(f"Entropie résiduelle (Standard) : {entropy_standard.item():.4f} (Chaos non régulé)")    print(f"Entropie résiduelle (Filtre SoS) : {entropy_sos.item():.4f} (Stabilité homéostatique)")        reduction = ((entropy_standard - entropy_sos) / entropy_standard) * 100    print(f"\n-> Dissipation native de l'anergie : -{reduction.item():.2f}% de bruit thermique.")    print("=========================================================")if __name__ == "__main__":    run_stability_test()
