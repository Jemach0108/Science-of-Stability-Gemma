import torchimport torch.nn as nnimport mathclass SoSAttentionModulator(nn.Module):"""Module d'ingénierie SoS (Science of Stability).Applique le Tenseur M_phi basé sur la suite de Fibonacci et le ratio d'orpour la dissipation native de l'anergie informationnelle."""def init(self, d_key, gamma=1.0):super(SoSAttentionModulator, self).init()self.d_key = d_keyself.gamma = gamma        # Définition mathématique de la constante d'or        self.phi = (1.0 + math.sqrt(5.0)) / 2.0                # Pré-calcul et mise en cache de la matrice M_phi pour optimiser les FLOPs        self.register_buffer('M_phi', self._build_harmonic_matrix())    def _build_harmonic_matrix(self):        # Génération de la matrice de Toeplitz harmonique invariante d'échelle        indices = torch.arange(self.d_key).float()        # Calcul de la matrice des distances absolues |i - j|        distance_matrix = torch.abs(indices.unsqueeze(0) - indices.unsqueeze(1))                # Application de la loi de décroissance de Fibonacci : M_phi = phi^(-delta * gamma)        harmonic_matrix = torch.pow(self.phi, -distance_matrix * self.gamma)        return harmonic_matrix    def forward(self, Q, K, V, mask=None):        """        Arguments:            Q: Tenseur de Requêtes [Batch_Size, Num_Heads, Seq_Len, d_key]            K: Tenseur de Clés    [Batch_Size, Num_Heads, Seq_Len, d_key]            V: Tenseur de Valeurs [Batch_Size, Num_Heads, Seq_Len, d_value]            mask: Masque optionnel (ex: causal pour l'auto-régression)        """        # 1. Calcul du produit scalaire classique des poids d'attention        # [Batch, Heads, Seq_Len, d_key] x [Batch, Heads, d_key, Seq_Len] -> [Batch, Heads, Seq_Len, Seq_Len]        scores = torch.matmul(Q, K.transpose(-2, -1)) / math.sqrt(self.d_key)                # 2. Injection de la Boussole SoS (Produit de Hadamard adaptatif)        # La matrice M_phi se diffuse automatiquement sur les dimensions Batch et Heads        scores_stables = scores * self.M_phi                # 3. Application du masque d'attention classique si présent        if mask is not None:            scores_stables = scores_stables.masked_fill(mask == 0, -1e9)                # 4. Normalisation par Softmax : transformation en distribution de probabilité pure        attention_probs = torch.softmax(scores_stables, dim=-1)                # 5. Extraction du signal à haute exergie        output = torch.matmul(attention_probs, V)                return output, attention_probs
